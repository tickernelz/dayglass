import log from "electron-log";
import { nativeImage, screen } from "electron";
export const selectDisplays = (displays, primaryId, displayIds, onlyPrimaryScreen) => {
    const wanted = (Array.isArray(displayIds) ? displayIds : [])
        .map((id) => String(id))
        .filter((id) => id.length > 0);
    const picked = wanted.length > 0 ? displays.filter((d) => wanted.includes(String(d.id))) : displays;
    if (!onlyPrimaryScreen || picked.length === 0)
        return picked;
    return [picked.find((d) => String(d.id) === String(primaryId)) ?? picked[0]];
};
export const displayRects = (displays, selected, width, height) => {
    const minX = Math.min(...displays.map((d) => d.bounds.x));
    const minY = Math.min(...displays.map((d) => d.bounds.y));
    const spanX = Math.max(...displays.map((d) => d.bounds.x + d.bounds.width)) - minX;
    const spanY = Math.max(...displays.map((d) => d.bounds.y + d.bounds.height)) - minY;
    const scale = width / spanX;
    if (!(scale > 0) || Math.abs(height / spanY - scale) > scale * 0.01)
        return null;
    return selected.map((d) => {
        const x = Math.round((d.bounds.x - minX) * scale);
        const y = Math.round((d.bounds.y - minY) * scale);
        return {
            x,
            y,
            width: Math.min(width, Math.round((d.bounds.x - minX + d.bounds.width) * scale)) - x,
            height: Math.min(height, Math.round((d.bounds.y - minY + d.bounds.height) * scale)) - y,
        };
    });
};
export const scopeToDisplays = (buffers, { onlyPrimaryScreen = false, displayIds = [] } = {}) => {
    const displays = screen.getAllDisplays();
    const selected = selectDisplays(displays, screen.getPrimaryDisplay().id, displayIds, onlyPrimaryScreen);
    if (selected.length === displays.length)
        return buffers;
    if (selected.length === 0) {
        log.warn(`display scope: none of the selected displays (${displayIds.join(",")}) is connected`);
        return null;
    }
    if (buffers.length !== 1) {
        log.warn(`display scope: ${buffers.length} frames cannot be mapped to displays`);
        return null;
    }
    const image = nativeImage.createFromBuffer(buffers[0]);
    const { width, height } = image.getSize();
    const rects = displayRects(displays, selected, width, height);
    if (!rects) {
        log.warn(`display scope: frame ${width}x${height} does not match the layout of ${displays.length} displays`);
        return null;
    }
    log.info(`display scope: cut ${width}x${height} frame to ${rects.map((r) => `${r.width}x${r.height}+${r.x}+${r.y}`).join(", ")}`);
    return rects.map((rect) => image.crop(rect).toPNG());
};
